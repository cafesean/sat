# Home Page & Authentication Flow Improvements

## Goals
1. Improve logged-out user experience
2. Secure vocab module behind authentication
3. Hide inactive modules (dashboard, reading/writing)
4. Show progress stats only when logged in

## Implementation Plan

### 1. Route Protection
- Add middleware to protect vocabulary routes
- Redirect unauthenticated users to login page
- Hide protected routes from navigation when logged out

```ts
// src/middleware.ts
import { auth } from '@/lib/auth'
import { NextResponse } from 'next/server'

export default auth((req) => {
  const protectedPaths = [
    '/vocabulary',
    '/dashboard',
    '/reading-writing'
  ]
  
  if (!req.auth && protectedPaths.some(path => 
    req.nextUrl.pathname.startsWith(path)
  ) {
    return NextResponse.redirect(
      new URL('/login', req.url)
    )
  }
})

export const config = {
  matcher: ['/((?!api|_next/static|_next/image|favicon.ico).*)']
}
```

### 2. Home Page Updates
- Conditionally render content based on auth status
- Show progress stats only when logged in
- Add clear CTAs for logged-out users

```tsx
// src/app/page.tsx
import { auth } from '@/lib/auth'

export default async function Home() {
  const session = await auth()
  
  return (
    <div className="container">
      {session ? (
        <AuthenticatedView user={session.user} />
      ) : (
        <PublicView />
      )}
    </div>
  )
}

function AuthenticatedView({ user }) {
  return (
    <>
      <h1>Welcome back, {user.name}</h1>
      <ProgressStats userId={user.id} />
      {/* ... */}
    </>
  )
}

function PublicView() {
  return (
    <>
      <h1>SAT Study App</h1>
      <p>Sign up to access vocabulary practice</p>
      <div className="flex gap-4">
        <Link href="/signup">Sign Up</Link>
        <Link href="/login">Login</Link>
      </div>
    </>
  )
}
```

### 3. Navigation Updates
- Hide protected routes from sidebar/navbar when logged out
- Update active state styling

```tsx
// src/components/sidebar.tsx
import { useSession } from 'next-auth/react'

export function Sidebar() {
  const { data: session } = useSession()
  
  return (
    <nav>
      {session && (
        <>
          <NavLink href="/dashboard">Dashboard</NavLink>
          <NavLink href="/vocabulary">Vocabulary</NavLink>
          <NavLink href="/reading-writing">Reading/Writing</NavLink>
        </>
      )}
      <NavLink href="/">Home</NavLink>
      {!session && (
        <>
          <NavLink href="/login">Login</NavLink>
          <NavLink href="/signup">Sign Up</NavLink>
        </>
      )}
    </nav>
  )
}
```

### 4. Progress Stats Component
- Fetch and display vocabulary practice stats
- Only show when authenticated

```tsx
// src/components/ProgressStats.tsx
'use client'

import { api } from '@/utils/trpc'

export function ProgressStats({ userId }) {
  const { data: stats } = api.vocab.getUserStats.useQuery({ userId })
  
  return (
    <div className="stats-grid">
      <StatCard 
        title="Words Learned" 
        value={stats?.wordsLearned} 
      />
      <StatCard
        title="Practice Sessions"
        value={stats?.sessionsCompleted}
      />
      <StatCard
        title="Retention Rate"
        value={`${stats?.retentionRate}%`}
      />
    </div>
  )
}
```

## Database Changes
- Add stats query to vocab router
- Create new procedure to fetch user stats

```ts
// src/server/api/routers/vocab.ts
const stats = protectedProcedure
  .input(z.object({ userId: z.string() }))
  .query(async ({ ctx, input }) => {
    return ctx.prisma.user.findUnique({
      where: { id: input.userId },
      select: {
        wordsLearned: true,
        sessionsCompleted: true,
        retentionRate: true
      }
    })
  })
```
